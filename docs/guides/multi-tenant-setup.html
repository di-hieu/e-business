Understanding Multi-Tenancy

What is Multi-Tenancy?

Multi-tenancy is an architectural pattern where a single instance of software serves multiple customers (tenants) while keeping their data logically isolated. This approach offers several advantages:

  • Cost Efficiency: Share infrastructure and reduce operational costs
  • Easier Maintenance: Update once, all tenants benefit
  • Scalability: Horizontal scaling serves all tenants simultaneously
  • Data Privacy: Tenant data is isolated and secure

Database-Level Isolation

In our POC phase, we use SQLite with tenant_id filtering. For production with PostgreSQL:

SQLite (POC)

# All tables share same database, filtered by tenant_id
CREATE TABLE conversations (
    id INTEGER PRIMARY KEY,
    tenant_id TEXT NOT NULL,
    user_id TEXT,
    message TEXT,
    response TEXT,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (tenant_id) REFERENCES tenants(id)

    id TEXT PRIMARY KEY,

);

-- All queries must include tenant filter
SELECT * FROM conversations WHERE tenant_id = ?;

Middleware for Tenant Isolation

Extract Tenant from Request

# FastAPI middleware to extract tenant ID
from fastapi import Request, Header, HTTPException
from fastapi.middleware.base import BaseHTTPMiddleware
from typing import Callable

async def tenant_middleware(request: Request, call_next: Callable) -> Response:
    # Try to extract tenant from header
    x_tenant_id = request.headers.get('X-Tenant-ID')
    authorization = request.headers.get('Authorization')
    
    if not x_tenant_id and not authorization:
        raise HTTPException(status_code=401, detail="Missing tenant identification")
    
    # Set current tenant in request scope
    request.state.tenant_id = x_tenant_id or extract_tenant_from_token(authorization)
    
    return await call_next(request)

# Add to main.py
app.add_middleware(TenantMiddleware)

# Or use Depends
from fastapi import Depends

async def get_current_tenant(
    x_tenant_id: str | None = Header(None),
    authorization: str | None = Header(None)
) -> Tenant:
    if not x_tenant_id and not authorization:
        raise HTTPException(status_code=401)
    
    tenant_id = x_tenant_id or extract_tenant_from_token(authorization)
    return await tenant_repository.get_tenant(tenant_id)

from sqlalchemy import column, literal, select, and_

class ConversationRepository:
    async def get_conversations(self, tenant: Tenant, limit: int = 50):
        # Inject tenant_id into query
        query = select(Conversation).where(

Rate Limiting per Tenant

Token Bucket Algorithm

# Per-tenant rate limiting with Redis
from aiolimiter import TokenBucket
import redis.asyncio as redis

class TenantRateLimiter:
    def __init__(self, default_quota: int = 1000, default_rate: int = 100):
        self.redis = redis.Redis(host='localhost', port=6379)
        self.default_quota = default_quota
        self.default_rate = default_rate
    
    async def get_or_create_bucket(self, tenant_id: str) -> TokenBucket:
        # Get tenant-specific quota from Redis
        config = await self.redis.get(f"tenant:{tenant_id}:config")
        if config:
            quota, rate = json.loads(config)
        else:
            quota, rate = self.default_quota, self.default_rate
        
        # Create tenant-specific bucket
        bucket_key = f"ratelimit:{tenant_id}"
        bucket = TokenBucket(quota, rate, redis_backend=self.redis)
        return bucket
    
    async def check_rate_limit(self, tenant_id: str) -> bool:
        bucket = await self.get_or_create_bucket(tenant_id)
        try:
            await bucket.acquire()
            return True
        except LimitExceeded:
            return False

Production Considerations

Scalability Limits

  • Memory: Monitor FAISS index size per tenant
  • Connection Pooling: Use PgBouncer for PostgreSQL
  • Database Size: Split large tenants to separate databases
  • Vector Search: Qdrant with partitioning for >1M vectors

Tenant Lifecycle

# Soft delete tenant (preserve data)
async def suspend_tenant(tenant_id: str):
    tenant = await get_tenant(tenant_id)
    tenant.is_suspended = True
    tenant.max_messages_per_day = 0
    
    # Disable rate limiting
    await rate_limiter.disable(tenant_id)
    
    # Keep data for 90 days then purge
    await schedule_purge(tenant_id, delay_days=90)

# Hard delete tenant (dangerous!)
async def delete_tenant(tenant_id: str):
    # Only for super admin
    tenant = await get_tenant(tenant_id)
    
    # Delete from all tables
    await database.delete_tenant_data(tenant_id)
    
    # Remove FAISS index
    faiss_path = f"{FAISS_DIR}/{tenant_id}.faiss"
    if os.path.exists(faiss_path):
        os.remove(faiss_path)
